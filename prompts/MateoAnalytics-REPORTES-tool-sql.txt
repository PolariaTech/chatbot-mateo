# Pegar en el nodo Postgres Tool que lista reportes.
# Cambio: filtra solo ambito = 'VENTA' y activos.

SET search_path TO "{{ $('Resolver Schema').first().json.schema }}";
SELECT c.embed_url, c.descripcion
FROM cuenta_reporte_embed c
INNER JOIN public.usuario u ON u.codigo_cuenta = c.codigo_cuenta
WHERE u.id_usuario = '{{ $('Preparar Payload').first().json.sessionId }}'
  AND c.esta_activo = true
  AND c.ambito = 'VENTA';
